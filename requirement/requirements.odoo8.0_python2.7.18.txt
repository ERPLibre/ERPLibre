# Priorité sur les requirements.txt d'Odoo 8 et des dépôts OCA.
#
# Poetry prend la DERNIÈRE version d'un paquet non borné, et beaucoup de
# paquets ont cessé Python 2 sans le déclarer dans leurs métadonnées : le
# solveur les retient, puis leur installation échoue. Chaque ligne ci-dessous
# est la dernière version dont les sources ou la roue tournent en 2.7.
paramiko==2.12.0
numpy==1.16.6
python-stdnum==1.20
suds==0.4
unidecode==1.2.0
click==7.1.2
geojson==2.5.0
shapely==1.7.1
evdev==1.5.0
py-Asterisk==0.5.18
# responses 0.10.6 passe encore par l'urllib3 que requests embarque : le
# requests épinglé par Odoo 8 et 9 n'installe pas urllib3 à part, et les
# versions suivantes l'importent directement.
responses==0.10.6
# La roue manylinux du psycopg2 qu'épingle Odoo embarque une libpq liée à
# libcrypt.so.1, que les distributions récentes ne livrent plus. La 2.8.6, dernière à
# accepter Python 2, n'a que ses sources sous Linux : elle se compile contre
# la libpq du système.
psycopg2==2.8.6
# num2words se déclare compatible 2.7 et emploie, dès la 0.5.13, une syntaxe
# réservée à Python 3.
num2words==0.5.12
# La 3.0 a cessé Python 2 : Poetry prendrait sinon la dernière, 3.x.
xlsxwriter==2.0.0
# Dernières versions qui tournent en 2.7, vérifiées par compilation : la
# plus récente de chacune a cessé Python 2, et « poetry add » la retiendrait.
acme_tiny==5.0.1
invoice2data==0.3.4
pycountry==20.7.3
pyotp==2.3.0
python-json-logger==0.1.11
# Dépendance indirecte : la version que Poetry retiendrait déclare
# accepter 2.7 mais ne compile que ses sources Python 3.
regex==2020.10.28
# La 2.0 est une extension C écrite pour Python 3 (PyModuleDef) ; la 1.9.73
# est celle que verrouille Odoo 10.
pycups==1.9.73
